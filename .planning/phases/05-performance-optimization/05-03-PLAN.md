    export interface PoolConfig<T> {
      initialSize: number;
      minSize: number;
      maxSize: number;
      factory: () => T;
      reset: (obj: T) => void; // Prepare for reuse
      onExpand?: (newSize: number) => void; // Logging callback
    }
    
    export interface PoolStats {
      totalAllocated: number;
      available: number;
      inUse: number;
      hitCount: number;
      missCount: number;
      expansionCount: number;
      hitRate: number; // 0-1
    }
    
    export class ObjectPool<T> {
      private available: T[] = [];
      private inUse = new Set<T>();
      private config: PoolConfig<T>;
      private stats = {
        hitCount: 0,
        missCount: 0,
        expansionCount: 0
      };
      
      constructor(config: PoolConfig<T>) {
        this.config = config;
        this.preAllocate(config.initialSize);
      }
      
      // Main API
      acquire(): T {
        if (this.available.length > 0) {
          this.stats.hitCount++;
          const obj = this.available.pop()!;
          this.inUse.add(obj);
          return obj;
        }
        
        this.stats.missCount++;
        
        // Auto-expand if under max
        if (this.inUse.size < this.config.maxSize) {
          this.expand();
          return this.acquire();
        }
        
        // Pool exhausted - create temporary (not tracked)
        console.warn('[ObjectPool] Pool exhausted, creating temporary object');
        return this.config.factory();
      }
      
      release(obj: T): void {
        if (!this.inUse.has(obj)) {
          console.warn('[ObjectPool] Releasing object not from pool');
          return;
        }
        
        this.inUse.delete(obj);
        this.config.reset(obj);
        
        // Don't exceed max, but keep at least min
        if (this.available.length < this.config.maxSize) {
          this.available.push(obj);
        }
      }
      
      // Statistics
      getStats(): PoolStats {
        const total = this.stats.hitCount + this.stats.missCount;
        return {
          totalAllocated: this.available.length + this.inUse.size,
          available: this.available.length,
          inUse: this.inUse.size,
          hitCount: this.stats.hitCount,
          missCount: this.stats.missCount,
          expansionCount: this.stats.expansionCount,
          hitRate: total > 0 ? this.stats.hitCount / total : 0
        };
      }
      
      // Reset stats (useful for benchmarking)
      resetStats(): void {
        this.stats = { hitCount: 0, missCount: 0, expansionCount: 0 };
      }
      
      // Clear pool (emergency only)
      clear(): void {
        // Reset all in-use objects
        for (const obj of this.inUse) {
          this.config.reset(obj);
        }
        this.inUse.clear();
        
        // Trim to min size
        this.available = this.available.slice(0, this.config.minSize);
      }
      
      private preAllocate(count: number): void {
        for (let i = 0; i < count; i++) {
          this.available.push(this.config.factory());
        }
      }
      
      private expand(): void {
        const currentSize = this.available.length + this.inUse.size;
        const expansionSize = Math.min(
          Math.max(10, Math.floor(currentSize * 0.2)), // 20% growth, min 10
          this.config.maxSize - currentSize
        );
        
        if (expansionSize > 0) {
          this.preAllocate(expansionSize);
          this.stats.expansionCount++;
          this.config.onExpand?.(currentSize + expansionSize);
        }
      }
    }

    export interface ChunkCoord {
      cx: number; // Chunk X
      cy: number; // Chunk Y
    }
    
    export interface Chunk {
      coord: ChunkCoord;
      tiles: TerrainTile[];
      entities: EntityData[];
      loaded: boolean;
      loading: boolean;
      lastAccessed: number; // Frame count
      memoryEstimate: number; // Bytes
    }
    
    export interface ChunkManagerConfig {
      chunkSize: number;        // Tiles per chunk dimension (default: 64)
      loadRadius: number;       // Chunks to load in each direction (default: 2 = 5×5)
      unloadDistance: number;   // Chunks away to unload (default: 8)
      maxConcurrentLoads: number; // Async load limit (default: 4)
    }
    
    export interface ChunkLoadResult {
      loaded: ChunkCoord[];
      unloaded: ChunkCoord[];
      loading: ChunkCoord[];
    }
    
    export class ChunkManager {
      private chunks = new Map<string, Chunk>(); // Key: "cx,cy"
      private config: ChunkManagerConfig;
      private frameCount = 0;
      private loadQueue: ChunkCoord[] = [];
      private loading = new Set<string>();
      
      constructor(config?: Partial<ChunkManagerConfig>) {
        this.config = {
          chunkSize: 64,
          loadRadius: 2,
          unloadDistance: 8,
          maxConcurrentLoads: 4,
          ...config
        };
      }
      
      // Coordinate conversion
      worldToChunk(worldX: number, worldY: number): ChunkCoord {
        return {
          cx: Math.floor(worldX / this.config.chunkSize),
          cy: Math.floor(worldY / this.config.chunkSize)
        };
      }
      
      chunkToWorld(coord: ChunkCoord): { x: number; y: number } {
        return {
          x: coord.cx * this.config.chunkSize,
          y: coord.cy * this.config.chunkSize
        };
      }
      
      // Main update - call each frame
      update(playerX: number, playerY: number): ChunkLoadResult {
        this.frameCount++;
        
        const playerChunk = this.worldToChunk(playerX, playerY);
        const result: ChunkLoadResult = { loaded: [], unloaded: [], loading: [] };
        
        // Determine which chunks should be loaded
        const desiredChunks = this.getChunksInRadius(playerChunk);
        
        // Unload distant chunks
        for (const [key, chunk] of this.chunks) {
          const distance = this.chunkDistance(chunk.coord, playerChunk);
          
          if (distance > this.config.unloadDistance) {
            this.unloadChunk(chunk);
            result.unloaded.push(chunk.coord);
          } else {
            // Update last accessed
            chunk.lastAccessed = this.frameCount;
          }
        }
        
        // Queue new chunks for loading
        for (const coord of desiredChunks) {
          const key = this.chunkKey(coord);
          if (!this.chunks.has(key) && !this.loading.has(key)) {
            this.loadQueue.push(coord);
          }
        }
        
        // Process load queue (async simulation)
        this.processLoadQueue(result);
        
        return result;
      }
      
      // Get chunk if loaded
      getChunk(coord: ChunkCoord): Chunk | undefined {
        return this.chunks.get(this.chunkKey(coord));
      }
      
      // Check if chunk is loaded
      isChunkLoaded(coord: ChunkCoord): boolean {
        const chunk = this.chunks.get(this.chunkKey(coord));
        return chunk?.loaded ?? false;
      }
      
      // Get all loaded chunks
      getLoadedChunks(): Chunk[] {
        return [...this.chunks.values()].filter(c => c.loaded);
      }
      
      // Statistics
      getStats(): {
        loadedCount: number;
        loadingCount: number;
        queueLength: number;
        totalMemoryEstimate: number;
      } {
        const loaded = this.getLoadedChunks();
        return {
          loadedCount: loaded.length,
          loadingCount: this.loading.size,
          queueLength: this.loadQueue.length,
          totalMemoryEstimate: loaded.reduce((sum, c) => sum + c.memoryEstimate, 0)
        };
      }
      
      // Force load a specific chunk (for testing/warmup)
      async forceLoadChunk(coord: ChunkCoord): Promise<Chunk> {
        const key = this.chunkKey(coord);
        
        if (this.chunks.has(key)) {
          return this.chunks.get(key)!;
        }
        
        const chunk = await this.loadChunkData(coord);
        this.chunks.set(key, chunk);
        return chunk;
      }
      
      // Clear all chunks
      clear(): void {
        this.chunks.clear();
        this.loadQueue = [];
        this.loading.clear();
      }
      
      private getChunksInRadius(center: ChunkCoord): ChunkCoord[] {
        const coords: ChunkCoord[] = [];
        const r = this.config.loadRadius;
        
        for (let dx = -r; dx <= r; dx++) {
          for (let dy = -r; dy <= r; dy++) {
            coords.push({ cx: center.cx + dx, cy: center.cy + dy });
          }
        }
        
        return coords;
      }
      
      private chunkDistance(a: ChunkCoord, b: ChunkCoord): number {
        return Math.max(Math.abs(a.cx - b.cx), Math.abs(a.cy - b.cy));
      }
      
      private chunkKey(coord: ChunkCoord): string {
        return `${coord.cx},${coord.cy}`;
      }
      
      private processLoadQueue(result: ChunkLoadResult): void {
        const toLoad = Math.min(
          this.loadQueue.length,
          this.config.maxConcurrentLoads - this.loading.size
        );
        
        for (let i = 0; i < toLoad; i++) {
          const coord = this.loadQueue.shift()!;
          const key = this.chunkKey(coord);
          
          this.loading.add(key);
          result.loading.push(coord);
          
          // Simulate async load
          this.loadChunkData(coord).then(chunk => {
            this.chunks.set(key, chunk);
            this.loading.delete(key);
            result.loaded.push(coord);
          });
        }
      }
      
      private async loadChunkData(coord: ChunkCoord): Promise<Chunk> {
        // In real implementation, this would load from disk or generate
        // For now, return empty chunk structure
        return {
          coord,
          tiles: [],
          entities: [],
          loaded: true,
          loading: false,
          lastAccessed: this.frameCount,
          memoryEstimate: 16384 // 16KB estimate
        };
      }
      
      private unloadChunk(chunk: Chunk): void {
        const key = this.chunkKey(chunk.coord);
        this.chunks.delete(key);
      }
    }
