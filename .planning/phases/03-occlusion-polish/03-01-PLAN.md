```typescript
export interface OcclusionAnimatorOptions {
  scene: Phaser.Scene;
  fadeDurationMs: number;
  targetAlpha: number;
}

export class OcclusionAnimator {
  private readonly tweens = new Map<string, Phaser.Tweens.Tween>();
  private readonly scene: Phaser.Scene;
  private readonly fadeDuration: number;
  private readonly targetAlpha: number;
  
  updateOcclusionState(occludedIds: string[], entitySprites: Map<string, EntitySprite>): void {
    // For each occluded entity: fade OUT to targetAlpha
    // For each non-occluded entity that was occluded: fade IN to 1.0
  }
  
  private fadeEntity(entity: EntitySprite, targetAlpha: number): void {
    const sprite = entity.sprite as Phaser.GameObjects.Image;
    const existingTween = this.tweens.get(entity.id);
    if (existingTween) existingTween.stop();
    
    const tween = this.scene.tweens.add({
      targets: sprite,
      alpha: targetAlpha,
      duration: this.fadeDuration,
      ease: 'Linear'
    });
    this.tweens.set(entity.id, tween);
  }
}
```

Implement height classification system that drives occlusion behavior per OCC-05 requirement.

**Classification Rules:**
```
flat:    renderHeight = 0,     no shadow, no occlusion
low:     renderHeight < 16,    small shadow, no occlusion
medium:  renderHeight 16-32,   medium shadow, canOccludePlayer configurable
 tall:    renderHeight > 32,    large shadow, canOccludePlayer = true
```

**Changes:**

1. **Add classification constants to occlusionManager.ts:**
   ```typescript
   export const HeightClassifications = {
     FLAT: { maxHeight: 0,    canOcclude: false, shadowSize: 'none' },
     LOW:  { maxHeight: 16,   canOcclude: false, shadowSize: 'small' },
     MEDIUM: { maxHeight: 32, canOcclude: 'configurable', shadowSize: 'medium' },
     TALL: { maxHeight: Infinity, canOcclude: true, shadowSize: 'large' }
   } as const;
   
   export function classifyHeight(renderHeight: number): HeightClassification {
     if (renderHeight <= 0) return 'flat';
     if (renderHeight < 16) return 'low';
     if (renderHeight < 32) return 'medium';
     return 'tall';
   }
   ```

2. **Auto-derive canOccludePlayer:**
   ```typescript
   export function shouldOccludePlayer(
     classification: HeightClassification,
     explicitCanOcclude?: boolean
   ): boolean {
     if (classification === 'tall') return true;
     if (classification === 'flat' || classification === 'low') return false;
     return explicitCanOcclude ?? false; // medium: use explicit or default false
   }
   ```

3. **Update GameViewport.getVisualPack() to set classification:**
   - Derive classification from renderHeight
   - Set canOccludePlayer based on classification rules
   - Add warning logging for misconfiguration

4. **Add validation logging:**
