import type { WorldX, WorldY } from "./coordinates";
import type { EntityType } from "./depthSorter";

/**
 * Individual layer within a layered entity
 * Each layer renders independently with its own depth
 */
export interface EntityLayer {
  id: string;
  renderHeight: number;
  frame: number;
  canOcclude: boolean;
  depth: number;
  sprite: unknown; // Framework-agnostic, Phaser.Image in practice
}

/**
 * Layered entity extending base EntitySprite
 * Backward compatible: if layers is undefined/empty, renders as single entity
 */
export interface LayeredEntity {
  id: string;
  x: WorldX;
  y: WorldY;
  baseRenderHeight: number;  // Sum of all layer heights
  type: EntityType;
  layers: EntityLayer[];
  needsDepthUpdate: boolean;
  heightClassification?: "flat" | "low" | "medium" | "tall";
  
  // Cached bounds for occlusion detection
  bounds: {
    minX: number;
    minY: number;
    maxX: number;
    maxY: number;
  };
}

/**
 * Configuration for creating a layered entity
 */
export interface LayeredEntityConfig {
  id: string;
  x: WorldX;
  y: WorldY;
  type: EntityType;
  heightClassification?: "flat" | "low" | "medium" | "tall";
  layers: LayerConfig[];
}

export interface LayerConfig {
  id: string;
  renderHeight: number;
  frame: number;
  canOcclude?: boolean; // default: true for top layers, false for bottom
}

/**
 * Type guard to check if an entity is layered
 */
export function isLayeredEntity(entity: unknown): entity is LayeredEntity {
  return entity != null && 
         typeof entity === 'object' && 
         'layers' in entity && 
         Array.isArray((entity as LayeredEntity).layers) &&
         (entity as LayeredEntity).layers.length > 0;
}

/**
 * Visual Pack v2 schema for pseudo-3D rendering
 * Extends v1 with height metadata, occlusion rules, and layer support
 */
export interface VisualPackV2 {
  /** Schema version - must be 2 for v2 features */
  visualPackVersion: 2;
  
  /** Content ID this visual pack applies to */
  contentId: string;
  
  /** Total visual height in pixels (sum of layer heights if layered) */
  renderHeight: number;
  
  /** Height classification for shadow/occlusion behavior */
  heightClassification: "flat" | "low" | "medium" | "tall";
  
  /** Footprint in tiles for spatial queries */
  footprint: {
    widthTiles: number;
    depthTiles: number;
  };
  
  /** Whether this entity can occlude the player (default based on heightClassification) */
  canOccludePlayer?: boolean;
  
  /** Target alpha when occluding player (0.0-1.0, default 0.4) */
  occlusionAlpha?: number;
  
  /** Layer definitions for split-layer entities (optional) */
  layers?: VisualPackLayer[];
}

/**
 * Individual layer definition within a Visual Pack v2
 */
export interface VisualPackLayer {
  /** Layer identifier (e.g., "trunk", "canopy") */
  id: string;
  
  /** Height of this layer in pixels */
  renderHeight: number;
  
  /** Sprite frame index for this layer */
  frame: number;
  
  /** Whether this layer can occlude the player */
  canOcclude?: boolean;
}

/**
 * Visual Pack v1 (legacy, backward compatible)
 */
export interface VisualPackV1 {
  contentId: string;
  // v1 has no formal schema - just frame mappings
  // This type exists for documentation/completeness
}

/**
 * Discriminated union of all visual pack versions
 */
export type VisualPack = VisualPackV1 | VisualPackV2;

/**
 * Type guard to detect v2 visual packs
 */
export function isVisualPackV2(pack: unknown): pack is VisualPackV2 {
  return pack != null && 
         typeof pack === 'object' && 
         'visualPackVersion' in pack && 
         (pack as VisualPackV2).visualPackVersion === 2;
}

/**
 * Get effective canOccludePlayer value
 * Uses height classification defaults if not explicitly set
 */
export function getCanOccludePlayer(
  pack: VisualPackV2
): boolean {
  if (pack.canOccludePlayer !== undefined) {
    return pack.canOccludePlayer;
  }
  // Default based on height classification
  return pack.heightClassification === 'tall';
}

/**
 * Get effective occlusion alpha
 */
export function getOcclusionAlpha(pack: VisualPackV2): number {
  return pack.occlusionAlpha ?? 0.4;
}

import type { WorldX, WorldY } from "./coordinates";
import type { LayeredEntity, EntityLayer } from "./layeredEntity";
import { TYPE_PRIORITY, DEPTH_POSITION_MULTIPLIER, type EntityType } from "./depthSorter";

/**
 * Options for layer depth calculation
 */
export interface LayerDepthOptions {
  baseX: WorldX;
  baseY: WorldY;
  baseOffset?: number;
}

/**
 * Result of layer depth calculation
 */
export interface CalculatedLayer {
  layer: EntityLayer;
  depth: number;
  worldY: WorldY; // The Y position this layer occupies for depth sorting
}

/**
 * Renderer for layered entities
 * Handles depth calculation and occlusion detection per-layer
 */
export class LayeredEntityRenderer {
  /**
   * Calculate depth for each layer of a layered entity
   * Layers stack upward from base position
   */
  calculateLayerDepths(
    entity: LayeredEntity,
    baseOffset: number = 0
  ): CalculatedLayer[] {
    const results: CalculatedLayer[] = [];
    let currentY = entity.y;
    
    for (const layer of entity.layers) {
      // Layer's effective Y is base Y minus cumulative height of layers below
      // This makes upper layers render in front (higher depth value)
      const layerY = (currentY - layer.renderHeight) as WorldY;
      
      // Depth uses same formula as regular entities
      const depth = (layerY + layer.renderHeight) * DEPTH_POSITION_MULTIPLIER + 
                    TYPE_PRIORITY[entity.type];
      
      results.push({
        layer,
        depth: depth + baseOffset,
        worldY: layerY
      });
      
      // Move up for next layer
      currentY -= layer.renderHeight;
    }
    
    return results;
  }
  
  /**
   * Check if a specific layer is occluding a position
   */
  isLayerOccludingPosition(
    entity: LayeredEntity,
    layerIndex: number,
    playerX: WorldX,
    playerY: WorldY
  ): boolean {
    const layer = entity.layers[layerIndex];
    if (!layer || !layer.canOcclude) {
      return false;
    }
    
    // Calculate cumulative height up to and including this layer
    let layerBaseY = entity.y;
    for (let i = 0; i < layerIndex; i++) {
      layerBaseY -= entity.layers[i].renderHeight;
    }
    const layerTopY = layerBaseY - layer.renderHeight;
    
    // Player is occluded by this layer if:
    // 1. Player is within entity footprint
    // 2. Player Y is between layer base and layer top
    const withinFootprint = this.isWithinFootprint(entity, playerX, playerY);
    const withinLayerHeight = playerY >= layerTopY && playerY <= layerBaseY;
    
    return withinFootprint && withinLayerHeight;
  }
  
  /**
   * Check which layers (if any) are occluding the player
   * Returns array of layer indices that should fade
   */
  getOccludingLayers(
    entity: LayeredEntity,
    playerX: WorldX,
    playerY: WorldY
  ): number[] {
    const occluding: number[] = [];
    
    for (let i = 0; i < entity.layers.length; i++) {
      if (this.isLayerOccludingPosition(entity, i, playerX, playerY)) {
        occluding.push(i);
      }
    }
    
    return occluding;
  }
  
  /**
   * Get total render height (sum of all layers)
   */
  getTotalHeight(entity: LayeredEntity): number {
    return entity.layers.reduce((sum, layer) => sum + layer.renderHeight, 0);
  }
  
  /**
   * Get the layer at a specific world Y position
   * Returns layer index or -1 if outside entity bounds
   */
  getLayerAtPosition(
    entity: LayeredEntity,
    worldY: WorldY
  ): number {
    let currentY = entity.y;
    
    for (let i = 0; i < entity.layers.length; i++) {
      const layerTopY = currentY - entity.layers[i].renderHeight;
      if (worldY >= layerTopY && worldY <= currentY) {
        return i;
      }
      currentY = layerTopY;
    }
    
    return -1;
  }
  
  private isWithinFootprint(
    entity: LayeredEntity,
    x: WorldX,
    y: WorldY
  ): boolean {
    return x >= entity.bounds.minX && 
           x <= entity.bounds.maxX &&
           y >= entity.bounds.minY && 
           y <= entity.bounds.maxY;
  }
}
```

**Key Behaviors:**
1. Layers stack upward - lower layers are at base Y, upper layers at lower Y values
2. Each layer gets independent depth calculation
3. Occlusion checked per-layer using cumulative height
4. Player between trunk and canopy triggers canopy fade only

import type { LayeredEntity } from "./layeredEntity";
import { LayeredEntityRenderer } from "./layeredEntityRenderer";

// Add to OcclusionCheckResult interface:
export interface OcclusionCheckResult {
  occludedEntities: string[];
  occludedLayers: Map<string, number[]>; // entityId -> layer indices that should fade
}

// Add to OcclusionManager class:
export class OcclusionManager {
  private readonly layeredRenderer = new LayeredEntityRenderer();
  private readonly occludedLayers = new Map<string, number[]>();
  
  checkOcclusion(
    playerX: WorldX,
    playerY: WorldY,
    entities: Map<string, EntitySprite | LayeredEntity>
  ): OcclusionCheckResult {
    const occludedEntities: string[] = [];
    this.occludedLayers.clear();
    
    for (const [id, entity] of entities) {
      if (isLayeredEntity(entity)) {
        // Check per-layer occlusion
        const occludingLayers = this.layeredRenderer.getOccludingLayers(
          entity, playerX, playerY
        );
        
        if (occludingLayers.length > 0) {
          occludedEntities.push(id);
          this.occludedLayers.set(id, occludingLayers);
        }
      } else {
        // Legacy single-entity occlusion
        if (this.isEntityOccludingPlayer(entity, playerX, playerY)) {
          occludedEntities.push(id);
        }
      }
    }
    
    return {
      occludedEntities,
      occludedLayers: new Map(this.occludedLayers)
    };
  }
  
  /**
   * Get which layers of an entity should fade (if any)
   */
  getOccludedLayers(entityId: string): number[] {
    return this.occludedLayers.get(entityId) ?? [];
  }
  
  /**
   * Check if a specific layer of a layered entity is occluded
   */
  isLayerOccluded(entityId: string, layerIndex: number): boolean {
    const layers = this.occludedLayers.get(entityId);
    return layers?.includes(layerIndex) ?? false;
  }
}
```

   private updateLayeredEntityAlpha(
     entity: LayeredEntity,
     occludedLayers: number[]
   ): void {
     for (let i = 0; i < entity.layers.length; i++) {
       const layer = entity.layers[i];
       const sprite = layer.sprite as Phaser.GameObjects.Image;
       const isOccluded = occludedLayers.includes(i);
       const targetAlpha = isOccluded ? (this.occlusionAlpha ?? 0.4) : 1.0;
       
       // Animate to target alpha

     sprite: Phaser.GameObjects.Image,
     targetAlpha: number
   ): void {
     // Stop existing tween if any
     const existingKey = `alpha_${sprite.name}`;
     this.scene.tweens.killTweensOf(sprite);
     
     // Create new tween
     this.scene.tweens.add({
       targets: sprite,
       alpha: targetAlpha,
       duration: 250,
       ease: 'Linear'
     });
   }

   private renderEntity(entity: EntitySprite | LayeredEntity): void {
     if (isLayeredEntity(entity)) {

     } else {
       this.renderSingleEntity(entity);
     }

     const calculated = this.layeredRenderer.calculateLayerDepths(entity, this.depthBaseOffset);
     
     for (let i = 0; i < entity.layers.length; i++) {
       const layer = entity.layers[i];
       const calc = calculated[i];
       const sprite = layer.sprite as Phaser.GameObjects.Image;
       
       // Update depth

       // Update position for visual stacking
       sprite.setY(entity.y - calc.layer.renderHeight / 2);
     }
   }
