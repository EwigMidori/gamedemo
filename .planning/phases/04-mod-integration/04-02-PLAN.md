       ```typescript
       export class ModTestHarness {
         private container: TestContainer;
         private loadedMods: Map<string, GameModModule> = new Map();
         private errors: Error[] = [];

         constructor() {
           this.container = createTestContainer();
         }

         async loadMod(modulePath: string): Promise<void> {
           const mod = await import(modulePath);
           const modModule = mod.default || mod[Object.keys(mod)[0]];
           
           try {
             await modModule.install(this.container);
             this.loadedMods.set(modModule.manifest.id, modModule);
           } catch (e) {
             this.errors.push(e as Error);
             throw e;
           }
         }

         getRegisteredContent(): ContentSnapshot {
           return this.container.content.snapshot();
         }

         getVisualPacks(): VisualPackMetadata[] {
           return this.container.content.getAllVisualPacks();
         }

         getErrors(): Error[] {
           return [...this.errors];
         }

         hasMod(modId: string): boolean {
           return this.loadedMods.has(modId);
         }
       }
       ```

       ```typescript
       export class ModCompatibilityTest {
         private harness: ModTestHarness;
         private results: ModTestResult[] = [];

         constructor() {
           this.harness = new ModTestHarness();
         }

         async runTest(modPath: string): Promise<ModTestResult> {
           const startTime = Date.now();
           const checks: CheckResult[] = [];
           const errors: string[] = [];

           // Check 1: Mod loads without errors
           try {
             await this.harness.loadMod(modPath);
             checks.push({ name: "mod-loads-without-errors", passed: true });
           } catch (e) {
             checks.push({
               name: "mod-loads-without-errors",
               passed: false,
               message: (e as Error).message
             });
             errors.push(`Load failed: ${(e as Error).message}`);
           }

           // Check 2: All visual packs valid
           const visualPacks = this.harness.getVisualPacks();
           const invalidPacks = visualPacks.filter(vp => {
             const warnings = validateVisualPack(vp as VisualPackV2);
             return warnings.length > 0;
           });
           
           if (invalidPacks.length === 0) {
             checks.push({ name: "visual-packs-valid", passed: true });
           } else {
             checks.push({
               name: "visual-packs-valid",
               passed: false,
               message: `${invalidPacks.length} invalid visual packs`
             });
           }

           // Check 3: Content registered
           const content = this.harness.getRegisteredContent();
           const hasContent = content.items.length > 0 || content.structures.length > 0;
           checks.push({
             name: "content-registered",
             passed: hasContent,
             message: hasContent ? undefined : "No items or structures registered"
           });

           // ... additional checks per MOD-05 checklist

           return {
             modId: this.extractModId(modPath),
             passed: checks.every(c => c.passed),
             checks,
             errors,
             duration: Date.now() - startTime
           };
         }

         async runAllTests(modPaths: string[]): Promise<ModTestResult[]> {
           this.results = [];
           for (const path of modPaths) {
             const result = await this.runTest(path);
             this.results.push(result);
           }
           return this.results;
         }

         generateReport(): string {
           const passed = this.results.filter(r => r.passed).length;
           const total = this.results.length;
           
           let report = `# Mod Compatibility Report\n\n`;
           report += `**Date:** ${new Date().toISOString()}\n`;
           report += `**Passed:** ${passed}/${total}\n\n`;
           
           for (const result of this.results) {
             report += `## ${result.modId}\n`;
             report += `Status: ${result.passed ? "✅ PASS" : "❌ FAIL"}\n`;
             report += `Duration: ${result.duration}ms\n\n`;
             
             for (const check of result.checks) {
               report += `- ${check.passed ? "✅" : "❌"} ${check.name}`;
               if (check.message) report += `: ${check.message}`;
               report += `\n`;
             }
             
             if (result.errors.length > 0) {
               report += `\n**Errors:**\n`;
               for (const error of result.errors) {
                 report += `- ${error}\n`;
               }
             }
             report += `\n`;
           }
           
           return report;
         }
       }
       ```

       ```typescript
       export const ModAssertions = {
         hasVisualPack(contentId: string, harness: ModTestHarness): void {
           const packs = harness.getVisualPacks();
           if (!packs.some(p => p.contentId === contentId)) {
             throw new Error(`Missing visual pack for ${contentId}`);
           }
         },

         hasHeightClassification(
           contentId: string,
           classification: HeightClassification,
           harness: ModTestHarness
         ): void {
           const pack = harness.getVisualPacks().find(p => p.contentId === contentId);
           if (!pack) {
             throw new Error(`Missing visual pack for ${contentId}`);
           }
           if (pack.heightClassification !== classification) {
             throw new Error(
               `${contentId} has classification ${pack.heightClassification}, expected ${classification}`
             );
           }
         }
       };
       ```
